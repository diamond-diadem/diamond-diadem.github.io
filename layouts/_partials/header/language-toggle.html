{{/* Segmented language switch (e.g. FR | EN). The current language is highlighted; every other language links to this page's translation or, when the page is untranslated and showMissingLanguages is on, to that language's home page. Context: dict with page - the current page class - extra CSS classes for the wrapper (string) */}} {{- $page := .page -}} {{- $translationURLs := dict -}} {{- range $page.Translations -}} {{- $translationURLs = merge $translationURLs (dict .Lang .RelPermalink) -}} {{- end -}}
{{- range hugo.Sites -}} {{- $lang := .Language.Lang -}} {{- $label := .Language.Label -}} {{- if eq $lang $page.Lang -}} {{ $label }} {{- else if not (in $page.Params.skipTranslations $lang) -}} {{- $translationURL := index $translationURLs $lang -}} {{- if $translationURL -}} {{ $label }} {{- else if site.Params.doks.showMissingLanguages -}} {{ $label }} {{- end -}} {{- end -}} {{- end -}}
