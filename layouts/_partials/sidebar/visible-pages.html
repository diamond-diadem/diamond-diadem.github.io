The section whose children to list. @returns {pages} */ -}} {{- $pages := .Pages.ByWeight -}} {{- if eq .Params.sidebar_sort "title" -}} {{- $pages = .Pages.ByLinkTitle -}} {{- end -}} {{- return where $pages "Params.sidebar_hidden" "ne" true -}}
